cmake_minimum_required(VERSION 3.20)
project(m_pocket VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lvgl" CACHE PATH "LVGL 9.5.0 source")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl EXCLUDE_FROM_ALL)
option(POCKET_BUILD_TESTS "Build host-side headless UI tests" OFF)
if(POCKET_BUILD_TESTS)
    add_executable(ui-test tests/ui_test.cpp)
    target_include_directories(ui-test PRIVATE vendor src)
    target_link_libraries(ui-test PRIVATE lvgl m)
    enable_testing()
    add_test(NAME ui COMMAND ui-test "${CMAKE_CURRENT_SOURCE_DIR}/src/service.py")
else()
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Device package must target Linux; use cmake/cardputer-toolchain.cmake")
    endif()
    add_executable(m-pocket src/ui.cpp)
    target_include_directories(m-pocket PRIVATE vendor)
    target_link_libraries(m-pocket PRIVATE lvgl m)
    target_compile_options(m-pocket PRIVATE -Wall -Wextra -Wpedantic -fPIE -fstack-protector-strong)
    target_link_options(m-pocket PRIVATE -pie -Wl,--as-needed,-z,relro,-z,now)
    install(TARGETS m-pocket RUNTIME DESTINATION bin)
    install(FILES src/service.py DESTINATION lib/m-pocket)
    install(FILES packaging/m-pocket.desktop DESTINATION share/APPLaunch/applications)
    install(FILES packaging/m-pocket.png DESTINATION share/APPLaunch/share/images)
    install(FILES README.md RELEASE-CHECK.md LICENSE THIRD-PARTY-NOTICES.txt DESTINATION share/doc/m-pocket)
endif()
